using CodeGenSample.Entities.Base;
using Nextended.Core.Attributes;

namespace CodeGenSample.Entities;

[AutoGenerateDto(ToDtoMethodName = "ToMegaDto", ToSourceMethodName = "AsSrc", IsComCompatible = true
    ,Namespace = "MyGeneration", GenerateBeforeAndAfterAssignPartialsInMapping = true 
)]
// every further [AutoGenerateDto] generates a DTO view: a flat (inherited members included) shape of its own, named by GeneratedClassName
[AutoGenerateDto(GeneratedClassName = "AddressLine", PropertiesToIgnore = ["Number", "ReferenceKey"])]
public class Address : EntityBase
{
    public string Street { get; set; }
    public string City { get; set; }
    public string Country { get; set; }
    public int Number { get; set; }
}
